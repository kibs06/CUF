# SoleVision — Product Reviews & Ratings Roadmap

**Feature:** Product Reviews & Ratings  
**Date:** July 16, 2026  
**Status:** Planning  

---

## Overview

Add a customer review and rating system to SoleVision. Customers who have purchased a product can leave a star rating (1–5) and written review. Sellers can view and respond. Reviews appear on product detail cards and product detail screens.

### 1.1 Create Migration File
**File:** `supabase/migrations/20260717_product_reviews.sql`

```sql
-- product_reviews table
CREATE TABLE IF NOT EXISTS public.product_reviews (
  id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  product_id      TEXT REFERENCES public.products(id) ON DELETE CASCADE,
  customer_id     UUID REFERENCES public.profiles(id) ON DELETE CASCADE,
  order_id        BIGINT REFERENCES public.orders(id) ON DELETE SET NULL,
  rating          INTEGER NOT NULL CHECK (rating >= 1 AND rating <= 5),
  title           TEXT,
  body            TEXT,
  is_verified     BOOLEAN DEFAULT false,
  created_at      TIMESTAMPTZ DEFAULT now()
);

-- One review per product per customer
CREATE UNIQUE INDEX idx_reviews_product_customer
  ON public.product_reviews(product_id, customer_id);

-- product_review_images table
CREATE TABLE IF NOT EXISTS public.product_review_images (
  id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  review_id       BIGINT REFERENCES public.product_reviews(id) ON DELETE CASCADE,
  image_url       TEXT NOT NULL,
  display_order   INTEGER DEFAULT 0
);

-- RLS Policies
ALTER TABLE public.product_reviews ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.product_review_images ENABLE ROW LEVEL SECURITY;

-- Anyone can read reviews
CREATE POLICY "Reviews are viewable by everyone"
  ON public.product_reviews FOR SELECT USING (true);

CREATE POLICY "Review images are viewable by everyone"
  ON public.product_review_images FOR SELECT USING (true);

-- Customers can insert reviews
CREATE POLICY "Customers can insert reviews"
  ON public.product_reviews FOR INSERT
  WITH CHECK (auth.uid() = customer_id);

-- Customers can update/delete own reviews
CREATE POLICY "Customers can update own reviews"
  ON public.product_reviews FOR UPDATE USING (auth.uid() = customer_id);

CREATE POLICY "Customers can delete own reviews"
  ON public.product_reviews FOR DELETE USING (auth.uid() = customer_id);

-- Review images follow review ownership
CREATE POLICY "Customers can manage their review images"
  ON public.product_review_images FOR ALL
  USING (
    EXISTS (
      SELECT 1 FROM public.product_reviews
      WHERE id = review_id AND customer_id = auth.uid()
    )
  );
```

### 1.2 Add Avg Rating to Products Table (Denormalized)
```sql
-- Add columns to products for fast reads
ALTER TABLE public.products
  ADD COLUMN IF NOT EXISTS avg_rating NUMERIC(2,1) DEFAULT 0,
  ADD COLUMN IF NOT EXISTS review_count INTEGER DEFAULT 0;

-- Trigger to auto-update avg_rating and review_count
CREATE OR REPLACE FUNCTION update_product_rating()
RETURNS TRIGGER AS $$
BEGIN
  UPDATE public.products SET
    avg_rating = COALESCE(
      (SELECT ROUND(AVG(rating)::numeric, 1) FROM public.product_reviews WHERE product_id = COALESCE(NEW.product_id, OLD.product_id)),
      0
    ),
    review_count = (
      SELECT COUNT(*) FROM public.product_reviews WHERE product_id = COALESCE(NEW.product_id, OLD.product_id)
    )
  WHERE id = COALESCE(NEW.product_id, OLD.product_id);
  RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE TRIGGER on_review_change
  AFTER INSERT OR UPDATE OR DELETE ON public.product_reviews
  FOR EACH ROW EXECUTE FUNCTION update_product_rating();
```

## Phase 2: Service Layer (Day 1–2)

### 2.1 Create Review Service
**File:** `lib/services/review_service.dart`

```dart
class ReviewService {
  ReviewService._();
  static final ReviewService instance = ReviewService._();

  SupabaseClient get _client => Supabase.instance.client;

  /// Get reviews for a product (paginated)
  Future<List<Map<String, dynamic>>> getReviews(String productId, {int offset = 0, int limit = 20}) async {
    return await _client
        .from('product_reviews')
        .select('*, profiles(full_name, avatar_url), product_review_images(*)')
        .eq('product_id', productId)
        .order('created_at', ascending: false)
        .range(offset, offset + limit - 1);
  }

  /// Get average rating for a product
  Future<Map<String, dynamic>> getRatingSummary(String productId) async {
    final data = await _client
        .from('products')
        .select('avg_rating, review_count')
        .eq('id', productId)
        .single();
    return data;
  }

  /// Check if current user has reviewed this product
  Future<bool> hasReviewed(String productId) async {
    final userId = _client.auth.currentUser?.id;
    if (userId == null) return false;
    final data = await _client
        .from('product_reviews')
        .select('id')
        .eq('product_id', productId)
        .eq('customer_id', userId)
        .maybeSingle();
    return data != null;
  }

  /// Submit a review. Throws if user already reviewed this product.
  Future<void> submitReview({
    required String productId,
    required int rating,
    String? title,
    String? body,
    String? orderId,
    List<XFile>? images,
  }) async {
    final userId = _client.auth.currentUser!.id;

    try {
      // Insert review
      final review = await _client
          .from('product_reviews')
          .insert({
            'product_id': productId,
            'customer_id': userId,
            'order_id': orderId,
            'rating': rating,
            'title': title,
            'body': body,
            'is_verified': orderId != null,
          })
          .select()
          .single();

      // Upload images if provided
      if (images != null && images.isNotEmpty) {
        final reviewId = review['id'] as int;
        for (int i = 0; i < images.length; i++) {
          final url = await _uploadReviewImage(userId, reviewId, images[i]);
          await _client.from('product_review_images').insert({
            'review_id': reviewId,
            'image_url': url,
            'display_order': i,
          });
        }
      }
    } on PostgrestException catch (e) {
      if (e.code == '23505') { // unique_violation
        throw Exception('You have already reviewed this product.');
      }
      rethrow;
    }
  }

  /// Delete a review
  Future<void> deleteReview(int reviewId) async {
    await _client.from('product_reviews').delete().eq('id', reviewId);
  }

  Future<String> _uploadReviewImage(
      String userId, int reviewId, XFile file) async {
    final bytes = await file.readAsBytes();
    final ext = file.path.split('.').last;
    // Uses 'avatars' bucket for now — consider creating a dedicated 'review-images' bucket
    final path = '$userId/reviews/$reviewId/${DateTime.now().millisecondsSinceEpoch}.$ext';
    await _client.storage.from('avatars').uploadBinary(path, bytes,
        fileOptions: FileOptions(contentType: 'image/$ext'));
    return _client.storage.from('avatars').getPublicUrl(path);
  }
}
```

### 2.2 Create Review Provider
**File:** `lib/providers/review_provider.dart`

```dart
class ReviewProvider extends ChangeNotifier {
  final ReviewService _service = ReviewService.instance;

  List<Map<String, dynamic>> _reviews = [];
  Map<String, dynamic> _ratingSummary = {'avg_rating': 0, 'review_count': 0};
  bool _isLoading = false;
  bool _hasReviewed = false;

  List<Map<String, dynamic>> get reviews => _reviews;
  Map<String, dynamic> get ratingSummary => _ratingSummary;
  bool get isLoading => _isLoading;
  bool get hasReviewed => _hasReviewed;

  Future<void> loadReviews(String productId) async {
    _isLoading = true;
    notifyListeners();
    try {
      _reviews = await _service.getReviews(productId);
      _ratingSummary = await _service.getRatingSummary(productId);
      _hasReviewed = await _service.hasReviewed(productId);
    } catch (_) {}
    _isLoading = false;
    notifyListeners();
  }

  Future<void> submitReview({
    required String productId,
    required int rating,
    String? title,
    String? body,
    String? orderId,
    List<XFile>? images,
  }) async {
    await _service.submitReview(
      productId: productId,
      rating: rating,
      title: title,
      body: body,
      orderId: orderId,
      images: images,
    );
    await loadReviews(productId); // refresh
  }
}
```
